/* Symmetrical 34px Topbar Icon Rail System */
.symmetrical-icon-rail {
  display: flex;
  align-items: center;
  gap: 6px;
}

.symmetrical-icon-rail .nav-icon {
  width: 34px;
  height: 34px;
  min-width: 34px;
  border-radius: 6px;
  display: inline-flex;
  align-items: center;
  justify-content: center;
  transition: all 0.22s cubic-bezier(0.16, 1, 0.3, 1);
}

.symmetrical-icon-rail .nav-icon svg {
  width: 16px;
  height: 16px;
  transition: transform 0.25s cubic-bezier(0.34, 1.56, 0.64, 1);
}

/* Ratchet gear step on hover */
.symmetrical-icon-rail .gear-btn:hover svg {
  transform: rotate(60deg);
}

/* Centrifuge momentum spin */
.symmetrical-icon-rail .refresh-btn:hover svg {
  animation: hudRadarSpin 1s infinite linear;
